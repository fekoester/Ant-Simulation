cmake_minimum_required(VERSION 3.16)
project(AntSimulation CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(ANTS_SANITIZE "Build with AddressSanitizer and UBSan" OFF)
if(ANTS_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(NOT MSVC)
  add_compile_options(-Wall -Wextra)
endif()

add_executable(ants src/main.cpp)
target_include_directories(ants PRIVATE include)
find_package(SFML 3 COMPONENTS Graphics QUIET)
if(SFML_FOUND)
  target_link_libraries(ants PRIVATE SFML::Graphics)
  message(STATUS "SFML 3 found: building the viewer")
else()
  find_package(SFML 2.5 COMPONENTS graphics window system QUIET)
  if(SFML_FOUND)
    target_link_libraries(ants PRIVATE sfml-graphics sfml-window sfml-system)
    message(STATUS "SFML 2 found: building the viewer")
  else()
    target_compile_definitions(ants PRIVATE ANTS_NO_SFML)
    message(STATUS "SFML not found: headless build only")
  endif()
endif()

enable_testing()
find_path(DOCTEST_INCLUDE doctest/doctest.h)
if(DOCTEST_INCLUDE)
  add_executable(ants_tests tests/test_ants.cpp)
  target_include_directories(ants_tests PRIVATE include ${DOCTEST_INCLUDE})
  add_test(NAME ants_tests COMMAND ants_tests)
endif()
